% Kryssruta
\newcommand{\kryssruta}[1]{%
  \begin{tikzpicture}[baseline=-0.5ex]
    \draw[thick] (0,0) rectangle (3.5mm,3.5mm);
    \node[anchor=west] at (4mm,1.75mm) {\small #1};
  \end{tikzpicture}%
}

% Kandidatrad med kryssruta
\newcommand{\kandidat}[4]{%
  \stepcounter{kandidatnr}%
  \noindent
  \begin{minipage}{\textwidth}
    \kryssruta{\thekanditatnr}%
    \hspace{2mm}%
    \textbf{#2 #3}%
    \ifthenelse{\equal{#4}{}}{}{, #4}%
  \end{minipage}%
  \par\vspace{1mm}%
}
